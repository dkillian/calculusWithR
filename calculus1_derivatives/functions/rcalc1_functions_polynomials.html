Exercise 2: Polynomials

Author

Dan Killian with Claude Sonnet 4.6

Published

September 3, 2026

Abstract
An R translation of Mike X Cohen, Calculus in Python

The Python version used SymPy to represent polynomial and rational expressions symbolically before plotting. In R, polynomials are evaluated numerically using vectorised arithmetic — no symbolic engine is needed. The polynom package exists for formal polynomial objects, but for plotting and evaluation a loop or vectorised expression is sufficient.

Exercise 1: Random polynomials

Generate a random degree-3 polynomial with standard-normal coefficients and plot it.

set.seed(983)   # allow different results each run

x <- seq(-5, 5, length.out = 20)

# Random coefficients (standard normal)
coefs <- rnorm(4)

# Build polynomial: sum c_i * x^i
y <- numeric(length(x))

for (i in seq_along(coefs)) {
  y <- y + coefs[i] * x^(i - 1)
}

out <- tibble(x=x, y=y) %>%
    round(1)

tbl_grob <- tableGrob(out,
                      rows=NULL,
                      theme = ttheme_minimal(base_size=12))

# Build a title string showing the polynomial
terms <- mapply(function(c, i) sprintf("%+.2f x^%d", c, i), coefs, 0:3)
title_str <- paste("y =", paste(terms, collapse = " "))

p <- ggplot(out, aes(x, y)) +
  geom_hline(yintercept = 0, color = "darkgoldenrod2", linewidth = 1, alpha = .5) +
  geom_vline(xintercept = 0, color = "darkgoldenrod2", linewidth = 1, alpha = .5) +
  geom_line(linewidth = 1.5, color = "dodgerblue2") +
  coord_cartesian(ylim = c(-15, 5)) +
  labs(x = "x", y = "f(x)", title = title_str) +
  theme(axis.title.y = element_text(angle = 0, vjust = .5))

grid.arrange(p, tbl_grob, ncol = 2, widths = c(3, 1))

Exercise 2: Rational function

\[f(x) = \frac{x^2 - 2x}{x^2 - 4}\]

In Python, SymPy plotted this symbolically. In R, we evaluate numerically and set values near the singularities (\(x = \pm 2\)) to NA — this prevents geom_line() from drawing a spurious vertical line across the asymptotes.

x <- seq(-3, 3, length.out = 60)

num <- x^2 - 2*x
den <- x^2 - 4

y <- num / den

# Mark values near singularities as NA to avoid vertical artifacts
y <- ifelse(abs(den) < 1e-6, NA, num / den)

out <- tibble(x=x,
              num=num,
              den=den,
              y=y) %>%
    round(1)

tbl_grob <- tableGrob(out,
                      rows=NULL,
                      theme = ttheme_minimal(base_size=12))

p <- ggplot(out, aes(x, y)) +
    geom_hline(yintercept = 0, color = "darkgoldenrod2", linewidth = 1, alpha = .5) +
    geom_vline(xintercept = 0, color = "darkgoldenrod2", linewidth = 1, alpha = .5) +
    geom_line(linewidth = 1, color = "dodgerblue2") +
    coord_cartesian(xlim = c(-3, 3), ylim = c(-20, 20)) +
    labs(x = "x", y = "f(x)",
       title = expression(f(x) == frac(x^2 - 2*x, x^2 - 4)),
       caption="Some sort of singularity at x = -2") +
    theme(axis.title.y = element_text(angle = 0, vjust = .5))

grid.arrange(p, tbl_grob, ncol = 2, widths = c(3, 1))

Exercise 3: Estimate a sine wave using polynomials

The Maclaurin series (Taylor series centered at 0) for sine:

\[\sin(x) \approx \sum_{n=1}^{N} \frac{(-1)^{n+1}\, x^{2n-1}}{(2n-1)!}\]

In Python, SymPy accumulated the series symbolically term by term. In R, we accumulate numerically — adding each term into a running total with a for loop. The comparison to the true sin(x) uses sampled points rather than a second line, making it easy to see where the approximation diverges near the edges of the domain.

Let’s start with \(n=1\)

n=1

order <- 10
x <- seq(-2*pi, 2*pi, length.out = 40)

sinx <- sin(x) %>% 
    round(4)

z <- numeric(length(x))

n <- 1
z <- z + (-1)^(n+1) * x^(2*n-1) / factorial(2*n-1)
y <- z

out <- tibble(x, sinx, y) %>%
    round(1)

tbl_grob <- tableGrob(out,
                      rows=NULL,
                      theme = ttheme_minimal(base_size=11))


p <- ggplot(out, aes(x,y)) +
    geom_hline(yintercept = 0, color = "darkgoldenrod2", linewidth = 1, alpha = .5) +
    geom_vline(xintercept = 0, color = "darkgoldenrod2", linewidth = 1, alpha = .5) +
    geom_point(color="dodgerblue2", size=2) +
    geom_line(color = "dodgerblue2", linewidth = 1) +
    geom_point(aes(y=sinx),
               size=.2,
               color="grey80") +
    geom_line(aes(y=sinx),
              size=.5,
              color="grey60",
              linetype="dashed") +
    coord_cartesian(xlim = c(min(x), max(x)),
                    ylim=c(-6.3, 6.3)) +
                    #ylim = c(floor(min(y) / 10) * 10, ceiling(max(y) / 10) * 10)) +
    scale_x_continuous(breaks=seq(round(min(x), -1), max(x), 1)) +
    scale_y_continuous(breaks=seq(round(min(y), -1), max(y), 1)) +
    labs(x = "x", y = "f(x)",
       title = paste("Maclaurin approximation of sin(x) for n = 1"),
       caption = "Line: Taylor series") + #     Points: sin(x)") +
    theme(axis.title.y = element_text(angle = 0, vjust = .5))

grid.arrange(p, tbl_grob, ncol = 2, widths = c(3, 1))

## Exercise 1: Random polynomials

Generate a random degree-3 polynomial with standard-normal coefficients and plot it.

```{r}
set.seed(983)   # allow different results each run

x <- seq(-5, 5, length.out = 20)

# Random coefficients (standard normal)
coefs <- rnorm(4)

# Build polynomial: sum c_i * x^i
y <- numeric(length(x))

for (i in seq_along(coefs)) {
```

## Exercise 2: Rational function

$$f(x) = \frac{x^2 - 2x}{x^2 - 4}$$

In Python, SymPy plotted this symbolically. In R, we evaluate numerically and set values near the singularities ($x = \pm 2$) to `NA` — this prevents `geom_line()` from drawing a spurious vertical line across the asymptotes.

```{r}
x <- seq(-3, 3, length.out = 60)

num <- x^2 - 2*x
den <- x^2 - 4

y <- num / den

# Mark values near singularities as NA to avoid vertical artifacts
y <- ifelse(abs(den) < 1e-6, NA, num / den)

out <- tibble(x=x,
              num=num,
              den=den,
              y=y) %>%
    round(1)

tbl_grob <- tableGrob(out,
                      rows=NULL,
                      theme = ttheme_minimal(base_size=12))

p <- ggplot(out, aes(x, y)) +
    geom_hline(yintercept = 0, color = "darkgoldenrod2", linewidth = 1, alpha = .5) +
    geom_vline(xintercept = 0, color = "darkgoldenrod2", linewidth = 1, alpha = .5) +
    geom_line(linewidth = 1, color = "dodgerblue2") +
    coord_cartesian(xlim = c(-3, 3), ylim = c(-20, 20)) +
```

## Exercise 3: Estimate a sine wave using polynomials

The Maclaurin series (Taylor series centered at 0) for sine:

$$\sin(x) \approx \sum_{n=1}^{N} \frac{(-1)^{n+1}\, x^{2n-1}}{(2n-1)!}$$

In Python, SymPy accumulated the series symbolically term by term. In R, we accumulate numerically — adding each term into a running total with a `for` loop. The comparison to the true `sin(x)` uses sampled points rather than a second line, making it easy to see where the approximation diverges near the edges of the domain.

Let's start with $n=1$

```{r}
x <- seq(-2*pi, 2*pi, length.out = 40)
sinx <- sin(x)
z <- numeric(length(x))

series <- matrix(NA, nrow = length(x), ncol = 9)

for (n in 1:9) {
  z <- z + (-1)^(n + 1) * x^(2*n - 1) / factorial(2*n - 1)
  series[, n] <- z
}

series <- as_tibble(series, .name_repair = ~ paste0("n=", 1:9)) |>
    mutate(x = x, sinx = sinx) |>
    relocate(x, sinx) 

seriesL <- series %>%
    pivot_longer(`n=1`:`n=9`, 
                 names_to="label",
                 values_to="y") %>%
```
